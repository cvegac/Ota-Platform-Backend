SELECT 
    u.username,
    r.role_name AS role_name,
    p.name AS permission_name
FROM
    admin_users.user u
        INNER JOIN
    admin_users.user_role ur ON u.id = ur.user_id
        INNER JOIN
    admin_users.role r ON ur.role_id = r.id
        INNER JOIN
    admin_users.role_permissions rp ON r.id = rp.role_id
        INNER JOIN
    admin_users.permission p ON rp.permission_id = p.id;
